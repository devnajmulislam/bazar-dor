import MarqueeText from "react-marquee-text";
import "react-marquee-text/dist/styles.css";

// Assuming you have your product data structure defined
interface IProduct {
  id: number;
  nameBn: string;
  categoryIcon: string;
  unit: string;
  today: number;
  change: {
    dir: string; // "up" or "down"
    pct: number;
  };
}

const ProductTicker = async () => {
  // 1. Server-side fetch data
  const res = await fetch(
    "https://api.api-store.workers.dev/api/bazardor/products",
    {
      cache: "no-store", // Good for real-time market data
    }
  );
  if (!res.ok) {
    throw new Error("Failed to fetch products");
  }
  const data: IProduct[] = await res.json();

  return (
    // 2. Main container and background
    <div className="bg-base-100 border-b border-base-200">
      <div className="container mx-auto">
        {/* 3. The Marquee Component */}
        {/* 'right' direction and 'loop' effect are key */}
        <MarqueeText 
          direction="right" 
          duration={30} // adjust speed as needed
          className="py-1" // minimal vertical padding
          pauseOnHover={true} // good user experience
        >
          {/* 4. Map data into repeating product blocks (cards) */}
          <div className="flex gap-1">
            {data.map((product) => {
              // Color-coding and arrow logic
              const isUp = product.change.dir === "up";
              const changeColor = isUp ? "text-error" : "text-success";
              const Arrow = isUp ? "▲" : "▼";

              return (
                <div
                  key={product.id}
                  // 5. Structure for distinct blocks with separators
                  // border-r and p-2 create the distinct "card" feel
                  className="flex items-center gap-2 p-2 border-r border-base-200"
                >
                  {/* Icon */}
                  <span className="text-lg opacity-70">
                    {product.categoryIcon}
                  </span>

                  {/* Product Name (Bangla) */}
                  <span className="font-medium text-base-content text-sm sm:text-base">
                    {product.nameBn}
                  </span>

                  {/* Price Block: "{today} টাকা/{unit}" */}
                  <div className="flex items-center gap-1.5 text-xs sm:text-sm text-base-content/80">
                    <span className="font-semibold text-base-content">
                      {product.today}
                    </span>
                    <span className="opacity-70">টাকা/</span>
                    <span className="opacity-70">
                      {product.unit}
                    </span>
                  </div>

                  {/* Change Percentage with Arrow and Color */}
                  <div className={`flex items-center gap-0.5 font-bold ${changeColor} text-xs sm:text-sm`}>
                    <span>{Arrow}</span>
                    <span>{product.change.pct.toFixed(1)}%</span>
                  </div>
                </div>
              );
            })}
          </div>
        </MarqueeText>
      </div>
    </div>
  );
};

export default ProductTicker;
